cmake_minimum_required(VERSION 3.20)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/../VERSION")
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/../VERSION" W11WIDGET_VERSION LIMIT_COUNT 1)
project(w11widget_native VERSION "${W11WIDGET_VERSION}" LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
enable_testing()

add_library(native_panel_core STATIC panel/PanelCore.cpp core/Configuration.cpp core/FocusTimer.cpp core/WidgetCatalog.cpp core/PopupPlacement.cpp
    ../src/core/PanelPositioner.cpp)
target_include_directories(native_panel_core PUBLIC panel core ../src)
add_executable(test_native_panel_core panel/test_panel_core.cpp)
target_link_libraries(test_native_panel_core PRIVATE native_panel_core)
add_test(NAME native_panel_core COMMAND test_native_panel_core)
foreach(component configuration focus_timer popup_placement)
    add_executable(test_native_${component} core/test_${component}.cpp)
    target_link_libraries(test_native_${component} PRIVATE native_panel_core)
    add_test(NAME native_${component} COMMAND test_native_${component})
endforeach()
add_executable(test_native_positioner ../tests/test_positioner.cpp)
target_link_libraries(test_native_positioner PRIVATE native_panel_core)
add_test(NAME native_positioner COMMAND test_native_positioner)

if(WIN32)
    enable_language(RC)
    configure_file(panel/Version.rc.in "${CMAKE_CURRENT_BINARY_DIR}/PanelVersion.rc" @ONLY)
    add_library(native_windows STATIC platform/SettingsStore.cpp platform/TimeZones.cpp
        ../src/platform/windows/TaskbarLocator.cpp)
    target_link_libraries(native_windows PUBLIC native_panel_core windowsapp shell32 ole32 icu)
    target_compile_definitions(native_windows PUBLIC UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
    add_executable(w11widget_native_panel WIN32 panel/main.cpp panel/ClockReadings.cpp
        panel/ProductionPanel.cpp panel/ShellIntegration.cpp)
    target_sources(w11widget_native_panel PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/PanelVersion.rc")
    target_compile_definitions(w11widget_native_panel PRIVATE W11WIDGET_VERSION="${PROJECT_VERSION}")
    target_link_libraries(w11widget_native_panel PRIVATE native_windows d2d1 dwrite dwmapi
        user32 gdi32 psapi advapi32 bcrypt comctl32)
    set_target_properties(w11widget_native_panel PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
    add_test(NAME native_panel_timezones COMMAND w11widget_native_panel --self-test)
    add_test(NAME native_smoke COMMAND w11widget_native_panel --smoke-test)
    add_executable(test_native_shell panel/test_shell_integration.cpp panel/ShellIntegration.cpp)
    target_link_libraries(test_native_shell PRIVATE native_windows bcrypt user32)
    add_test(NAME native_shell COMMAND test_native_shell)
    foreach(component settings_store timezones)
        add_executable(test_native_${component} platform/test_${component}.cpp)
        target_link_libraries(test_native_${component} PRIVATE native_windows)
        add_test(NAME native_${component} COMMAND test_native_${component})
    endforeach()
endif()

foreach(target native_panel_core native_windows test_native_panel_core test_native_configuration
    test_native_focus_timer test_native_popup_placement test_native_positioner test_native_settings_store test_native_timezones test_native_shell w11widget_native_panel)
    if(TARGET ${target})
        if(MSVC)
            set_property(TARGET ${target} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
            target_compile_options(${target} PRIVATE /W4 /permissive- /utf-8 /Zc:__cplusplus)
        else()
            target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
        endif()
    endif()
endforeach()
if(NOT WIN32)
    message(STATUS "Native panel HWND/Direct2D/ICU checks require Windows; building portable contract tests only.")
endif()
