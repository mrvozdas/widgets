cmake_minimum_required(VERSION 3.20)
project(w11widget_native_prototype VERSION 0.0.1 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
enable_testing()

add_library(native_panel_core STATIC panel/PanelCore.cpp ../src/core/PanelPositioner.cpp)
target_include_directories(native_panel_core PUBLIC panel ../src)
add_executable(test_native_panel_core panel/test_panel_core.cpp)
target_link_libraries(test_native_panel_core PRIVATE native_panel_core)
add_test(NAME native_panel_core COMMAND test_native_panel_core)

if(WIN32)
    add_executable(w11widget_native_panel WIN32 panel/main.cpp panel/ClockReadings.cpp)
    target_compile_definitions(w11widget_native_panel PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
    target_link_libraries(w11widget_native_panel PRIVATE native_panel_core d2d1 dwrite dwmapi
        user32 gdi32 shell32 ole32 psapi advapi32 icu)
    set_target_properties(w11widget_native_panel PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
    add_test(NAME native_panel_timezones COMMAND w11widget_native_panel --self-test)
endif()

foreach(target native_panel_core test_native_panel_core w11widget_native_panel)
    if(TARGET ${target})
        if(MSVC)
            set_property(TARGET ${target} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
            target_compile_options(${target} PRIVATE /W4 /permissive- /utf-8 /Zc:__cplusplus)
        else()
            target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
        endif()
    endif()
endforeach()
if(NOT WIN32)
    message(STATUS "Native panel HWND/Direct2D/ICU checks require Windows; building portable contract tests only.")
endif()
